## TECH STACK — USE EXACTLY THIS

**Frontend:** React + Vite + TypeScript + TailwindCSS  
**Backend:** Node.js + Express + TypeScript  
**Database:** SQLite with better-sqlite3 (no external DB needed — works natively in Replit)  
**Auth:** JWT + bcryptjs  
**Real-time:** Socket.io  
**AI (RakshBot):** Anthropic Claude API  
**File Upload:** Multer (local storage in Replit)  
**State:** Zustand  
**HTTP Client:** Axios  
**Icons:** Lucide React  
**Charts:** Recharts  
**Routing:** React Router v6  
**Notifications:** react-hot-toast

## DATABASE SCHEMA — CREATE ALL TABLES IN `server/src/database/init.ts`

Use better-sqlite3 (synchronous). Create all tables if they don't exist:

```sql
-- USERS
CREATE TABLE IF NOT EXISTS users (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  phone TEXT,
  password_hash TEXT NOT NULL,
  role TEXT CHECK(role IN ('student','police','company','admin')) DEFAULT 'student',
  college_name TEXT,
  college_city TEXT,
  state TEXT,
  student_id_url TEXT,
  skill_level TEXT DEFAULT 'beginner',
  skill_score INTEGER DEFAULT 0,
  total_points INTEGER DEFAULT 0,
  national_rank INTEGER,
  state_rank INTEGER,
  total_earned REAL DEFAULT 0,
  wallet_balance REAL DEFAULT 0,
  is_verified INTEGER DEFAULT 0,
  is_banned INTEGER DEFAULT 0,
  fraud_flag_count INTEGER DEFAULT 0,
  avatar_url TEXT,
  bio TEXT,
  github_url TEXT,
  linkedin_url TEXT,
  station_name TEXT,
  station_id TEXT,
  badge_number TEXT,
  jurisdiction_state TEXT,
  jurisdiction_city TEXT,
  company_name TEXT,
  company_website TEXT,
  gst_number TEXT,
  created_at TEXT DEFAULT (datetime('now')),
  updated_at TEXT DEFAULT (datetime('now')),
  last_login TEXT
);

-- SUBMISSIONS
CREATE TABLE IF NOT EXISTS submissions (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  title TEXT NOT NULL,
  description TEXT NOT NULL,
  type TEXT CHECK(type IN ('vulnerability','fraud_report','research','ctf_solution')) NOT NULL,
  severity TEXT CHECK(severity IN ('critical','high','medium','low','info')),
  category TEXT CHECK(category IN ('web','network','mobile','social_engineering','financial','hardware','malware','osint','other')),
  proof_files TEXT DEFAULT '[]',
  video_url TEXT,
  screenshots TEXT DEFAULT '[]',
  steps_to_reproduce TEXT,
  impact_description TEXT,
  fix_suggestion TEXT,
  target_url TEXT,
  affected_system TEXT,
  ai_plagiarism_score REAL DEFAULT 0,
  ai_analysis TEXT,
  status TEXT DEFAULT 'pending' CHECK(status IN ('pending','under_review','verified','rejected','fraud','escalated')),
  reviewer_id TEXT REFERENCES users(id),
  reviewer_notes TEXT,
  reviewed_at TEXT,
  assigned_department TEXT,
  escalated_to TEXT,
  points_awarded INTEGER DEFAULT 0,
  created_at TEXT DEFAULT (datetime('now')),
  updated_at TEXT DEFAULT (datetime('now'))
);

-- REWARDS
CREATE TABLE IF NOT EXISTS rewards (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  submission_id TEXT REFERENCES submissions(id),
  user_id TEXT REFERENCES users(id),
  amount REAL NOT NULL,
  currency TEXT DEFAULT 'INR',
  source TEXT CHECK(source IN ('government','corporate','platform','bounty')),
  source_org TEXT,
  status TEXT DEFAULT 'pending' CHECK(status IN ('pending','approved','paid','frozen','cancelled')),
  frozen_reason TEXT,
  payment_reference TEXT,
  approved_by TEXT REFERENCES users(id),
  approved_at TEXT,
  paid_at TEXT,
  created_at TEXT DEFAULT (datetime('now'))
);

-- BADGES
CREATE TABLE IF NOT EXISTS badges (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  name TEXT NOT NULL,
  description TEXT,
  icon TEXT,
  category TEXT,
  points_required INTEGER DEFAULT 0,
  is_rare INTEGER DEFAULT 0,
  color TEXT,
  emoji TEXT
);

-- USER_BADGES
CREATE TABLE IF NOT EXISTS user_badges (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  badge_id TEXT REFERENCES badges(id),
  earned_at TEXT DEFAULT (datetime('now')),
  UNIQUE(user_id, badge_id)
);

-- COURSES
CREATE TABLE IF NOT EXISTS courses (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  title TEXT NOT NULL,
  description TEXT,
  thumbnail_url TEXT,
  instructor_name TEXT,
  instructor_org TEXT,
  level TEXT,
  category TEXT,
  language TEXT DEFAULT 'English',
  duration_hours REAL,
  total_lessons INTEGER DEFAULT 0,
  is_free INTEGER DEFAULT 1,
  price REAL DEFAULT 0,
  is_certified INTEGER DEFAULT 0,
  enrolled_count INTEGER DEFAULT 0,
  rating REAL DEFAULT 4.5,
  tags TEXT DEFAULT '[]',
  created_at TEXT DEFAULT (datetime('now'))
);

-- COURSE_ENROLLMENTS
CREATE TABLE IF NOT EXISTS course_enrollments (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  course_id TEXT REFERENCES courses(id),
  progress_percent INTEGER DEFAULT 0,
  started_at TEXT DEFAULT (datetime('now')),
  completed_at TEXT,
  UNIQUE(user_id, course_id)
);

-- INTERNSHIPS
CREATE TABLE IF NOT EXISTS internships (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  org_id TEXT REFERENCES users(id),
  org_type TEXT CHECK(org_type IN ('government','private')),
  org_name TEXT NOT NULL,
  title TEXT NOT NULL,
  description TEXT,
  requirements TEXT,
  duration_months INTEGER,
  location_type TEXT CHECK(location_type IN ('remote','onsite','hybrid')),
  location_city TEXT,
  location_state TEXT,
  stipend_amount REAL,
  required_min_rank INTEGER,
  required_skill_score INTEGER,
  total_seats INTEGER DEFAULT 1,
  filled_seats INTEGER DEFAULT 0,
  status TEXT DEFAULT 'open' CHECK(status IN ('open','closed','filled')),
  application_deadline TEXT,
  start_date TEXT,
  perks TEXT DEFAULT '[]',
  is_govt_certified INTEGER DEFAULT 0,
  logo_url TEXT,
  created_at TEXT DEFAULT (datetime('now'))
);

-- INTERNSHIP_APPLICATIONS
CREATE TABLE IF NOT EXISTS internship_applications (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  internship_id TEXT REFERENCES internships(id),
  user_id TEXT REFERENCES users(id),
  cover_letter TEXT,
  status TEXT DEFAULT 'applied' CHECK(status IN ('applied','shortlisted','interview','selected','rejected')),
  offer_letter_url TEXT,
  applied_at TEXT DEFAULT (datetime('now')),
  UNIQUE(internship_id, user_id)
);

-- POLICE_CASES
CREATE TABLE IF NOT EXISTS police_cases (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  submission_id TEXT REFERENCES submissions(id),
  police_station_id TEXT REFERENCES users(id),
  assigned_officer_id TEXT REFERENCES users(id),
  case_number TEXT UNIQUE,
  status TEXT DEFAULT 'open' CHECK(status IN ('open','investigating','resolved','closed','escalated')),
  evidence_package_url TEXT,
  fir_number TEXT,
  notes TEXT,
  resolution_notes TEXT,
  priority TEXT DEFAULT 'medium' CHECK(priority IN ('critical','high','medium','low')),
  created_at TEXT DEFAULT (datetime('now')),
  updated_at TEXT DEFAULT (datetime('now')),
  resolved_at TEXT
);

-- CYBER_ALERTS
CREATE TABLE IF NOT EXISTS cyber_alerts (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  title TEXT NOT NULL,
  description TEXT,
  severity TEXT CHECK(severity IN ('critical','high','medium','low')),
  category TEXT,
  affected_states TEXT DEFAULT '[]',
  source TEXT,
  is_active INTEGER DEFAULT 1,
  is_verified INTEGER DEFAULT 0,
  reported_by TEXT REFERENCES users(id),
  created_at TEXT DEFAULT (datetime('now'))
);

-- CTF_CHALLENGES
CREATE TABLE IF NOT EXISTS ctf_challenges (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  title TEXT NOT NULL,
  description TEXT,
  category TEXT,
  difficulty TEXT CHECK(difficulty IN ('easy','medium','hard','expert')),
  points INTEGER DEFAULT 100,
  flag TEXT,
  hints TEXT DEFAULT '[]',
  attachment_url TEXT,
  is_active INTEGER DEFAULT 1,
  solve_count INTEGER DEFAULT 0,
  created_at TEXT DEFAULT (datetime('now'))
);

-- CTF_SOLUTIONS
CREATE TABLE IF NOT EXISTS ctf_solutions (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  challenge_id TEXT REFERENCES ctf_challenges(id),
  user_id TEXT REFERENCES users(id),
  is_correct INTEGER DEFAULT 0,
  time_taken_seconds INTEGER,
  solved_at TEXT DEFAULT (datetime('now')),
  UNIQUE(challenge_id, user_id)
);

-- WALLET_TRANSACTIONS
CREATE TABLE IF NOT EXISTS wallet_transactions (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  type TEXT CHECK(type IN ('credit','debit','withdrawal','bonus')),
  amount REAL NOT NULL,
  balance_after REAL,
  description TEXT,
  reference_id TEXT,
  status TEXT DEFAULT 'completed',
  created_at TEXT DEFAULT (datetime('now'))
);

-- BUG_BOUNTY_PROGRAMS
CREATE TABLE IF NOT EXISTS bug_bounty_programs (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  company_id TEXT REFERENCES users(id),
  company_name TEXT NOT NULL,
  title TEXT NOT NULL,
  description TEXT,
  scope TEXT,
  out_of_scope TEXT,
  min_reward REAL,
  max_reward REAL,
  status TEXT DEFAULT 'active',
  total_paid REAL DEFAULT 0,
  total_reports INTEGER DEFAULT 0,
  logo_url TEXT,
  created_at TEXT DEFAULT (datetime('now'))
);

-- NOTIFICATIONS
CREATE TABLE IF NOT EXISTS notifications (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  title TEXT NOT NULL,
  message TEXT,
  type TEXT,
  is_read INTEGER DEFAULT 0,
  action_url TEXT,
  created_at TEXT DEFAULT (datetime('now'))
);

-- CHAT_MESSAGES (RakshBot history per user)
CREATE TABLE IF NOT EXISTS chat_messages (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  user_id TEXT REFERENCES users(id),
  role TEXT CHECK(role IN ('user','assistant')),
  content TEXT NOT NULL,
  created_at TEXT DEFAULT (datetime('now'))
);
```
